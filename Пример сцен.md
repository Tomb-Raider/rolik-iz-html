# Пример сцен

Так выглядит правильный ответ ИИ по инструкции. Этот же код уже стоит в `Плеер.html`, его видно по кнопке «Код» → «Показать весь код»

Три сцены на 8 секунд
- «Титул» название набирается по букве, каретка мигает от времени, а не CSS-анимацией
- «Фразы» три строки по пресету «Короткие фразы как в рекламе Apple»
- «Цвета» цвета бренда из переменных, точки встают по одной и сливаются в полосу

Шрифт в примере системный (на Mac это SF Pro, на Windows Segoe UI). Чтобы поставить свой, впишите его название первым в `--font`, если шрифт установлен на компьютере

```js
// Пример ролика на три сцены. Так выглядит ответ ИИ по инструкции

reel({
  title: 'Ролик из HTML',
  fps: 24,
  css: `
    :root {
      --bg: #0B0D10;
      --fg: #FFFFFF;
      --accent: #3D7CFF;
      --c1: #3D7CFF; --c2: #22C1A6; --c3: #FFC53D; --c4: #FF6B4A; --c5: #B26BFF;
      /* Свой шрифт. Если он установлен на компьютере, впишите его название первым,
         например "SB Sans Display", system-ui, sans-serif */
      --font: system-ui, -apple-system, "Segoe UI", Roboto, Arial, sans-serif;
    }
  `
})

// Сцена 1. Название набирается по букве. Даже каретка мигает от времени t, а не CSS-анимацией
scene({
  name: 'Титул',
  dur: 2,
  note: 'Название набирается по букве, каретка цвета акцента',
  html: `<div class="title"><span class="txt"></span><i class="caret"></i></div>`,
  css: `
    .title { position: absolute; inset: 0; display: flex; align-items: center; justify-content: center;
             font-weight: 700; font-size: 154px; letter-spacing: -0.02em; white-space: pre; }
    .caret { display: block; width: 0.07em; height: 0.78em; margin-left: 0.05em; background: var(--accent); }
  `,
  draw({ t, $ }) {
    const text = 'Ролик из HTML'
    const n = Math.min(text.length, Math.floor(R.snap(t, 24) / 0.07))    // буква каждые 70 мс
    $('.txt').textContent = text.slice(0, n)
    const blink = Math.floor(t / 0.5) % 2 === 0                           // мигание раз в 0,5 с
    $('.caret').style.visibility = n < text.length || blink ? 'visible' : 'hidden'
    const s = 1 - 0.03 * R.ease.inOut(R.prog(t, 1.2, 0.8))               // лёгкий отъезд в конце
    $('.title').style.transform = `scale(${s})`
  }
})

// Пресет «короткие фразы как в рекламе Apple». Числа сняты покадрово с референса
const FPS = 24, F = 1 / FPS

function phrases(name, lines, { life = 1.2, first = 0.43, enter = true } = {}) {
  return {
    name,
    dur: lines.length * life,
    note: lines.map(l => l.join(' ')).join(' / '),
    html: lines.map(([a, b]) =>
      `<div class="line"><span>${a}</span><span class="add"><span>&nbsp;${b}</span></span></div>`).join(''),
    css: `
      .line { position: absolute; inset: 0; display: flex; align-items: center; justify-content: center;
              font-weight: 700; font-size: 154px; line-height: 1.25; letter-spacing: -0.015em; white-space: nowrap; }
      .add { display: flex; justify-content: flex-end; overflow: hidden; }
      .add > span { flex: none; }
    `,
    draw({ t, $$, w }) {
      t = R.snap(t, FPS)                                          // рубленая сетка 24 кадра в секунду
      const k = Math.min(lines.length - 1, Math.floor(t / life))  // какая строка сейчас
      $$('.line').forEach((line, i) => {
        line.style.display = i === k ? '' : 'none'               // склейка: кадр старой строки, следующий кадр новой
        if (i !== k) return
        const lt = t - k * life                                   // время внутри строки
        const add = line.lastElementChild, word = add.firstElementChild
        add.style.width = word.offsetWidth * R.ease.inOutQuint(R.prog(lt, first, 0.25)) + 'px'  // дописывание 250 мс
        word.style.visibility = lt >= first + 0.167 ? 'visible' : 'hidden'  // слово возникает за 83 мс до остановки
        let s = 1 - 0.15 * lt                                     // непрерывный отъезд, -15 % в секунду
        if (lt >= life - 2 * F) s -= lt >= life - F ? 0.03 : 0.01 // разгон перед склейкой
        if (k > 0 || !enter) s *= 1 + 0.05 * (1 - R.ease.out(R.prog(lt, 0, 0.06)))  // удар +5 % на склейке
        const x = k === 0 && enter ? -0.15 * w * (1 - R.ease.enter(R.prog(lt, 0, 0.417))) : 0  // въезд слева
        line.style.transform = `translateX(${x}px) scale(${s})`
      })
    }
  }
}

// Сцена 2. Три строки по пресету
scene(phrases('Фразы', [
  ['Текст', 'настоящий.'],
  ['Цвет', 'из бренда.'],
  ['Правка', 'за минуту.'],
]))

// Сцена 3. Цвета бренда из переменных: точки встают по одной и собираются в полосу
scene({
  name: 'Цвета',
  dur: 2.4,
  note: 'Пять точек цветов бренда встают по одной и сливаются в полосу, снизу подпись',
  html: `
    <div class="row">${[1, 2, 3, 4, 5].map(i => `<i style="background: var(--c${i})"></i>`).join('')}</div>
    <p class="cap">Ваш шрифт и ваши цвета</p>`,
  css: `
    .row { position: absolute; left: 50%; top: 44%; display: flex; transform: translate(-50%, -50%); }
    .row i { display: block; height: 120px; }
    .cap { position: absolute; left: 0; right: 0; top: 60%; margin: 0; text-align: center;
           font-size: 64px; font-weight: 600; letter-spacing: -0.01em; }
  `,
  draw({ t, $, $$ }) {
    t = R.snap(t, 24)
    const join = R.ease.inOutQuint(R.prog(t, 1.25, 0.5))             // точки сливаются в полосу
    $$('.row i').forEach((dot, i) => {
      const k = R.ease.out(R.prog(t, i * 0.2, 0.25))                  // точка встаёт за 0,25 с, по одной
      const r = R.mix(60, 0, join)                                    // скругление между точками уходит
      dot.style.width = R.mix(120, 200, join) + 'px'
      dot.style.marginLeft = i ? R.mix(40, 0, join) + 'px' : '0'
      dot.style.borderRadius = `${i === 0 ? 60 : r}px ${i === 4 ? 60 : r}px ${i === 4 ? 60 : r}px ${i === 0 ? 60 : r}px`
      dot.style.transform = `scale(${k})`
    })
    $('.cap').style.visibility = t >= 1.8 ? 'visible' : 'hidden'     // подпись встаёт склейкой, без проявления
  }
})
```
